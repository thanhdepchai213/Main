CheckMoon = function()
	local MoonTextureId = game:GetService("Lighting").Sky.MoonTextureId
	local s1 = "Bad Moon"

	if MoonTextureId == "http://www.roblox.com/asset/?id=9709149431" or MoonTextureId == "http://www.roblox.com/asset/?id=9709149052" then
		if MoonTextureId == "http://www.roblox.com/asset/?id=9709149431" then
			s1 = "Full Moon"
		elseif MoonTextureId == "http://www.roblox.com/asset/?id=9709149052" then
			s1 = "Next Night"
		end
	end

	return s1
end
function6 = function()
	return math.floor(game.Lighting.ClockTime)
end
function8 = function()
	local ClockTime = game.Lighting.ClockTime

	if CheckMoon() == "Full Moon" and ClockTime <= 5 then
		return tostring(function6()) .. " ( Will End Moon In " .. math.floor(5 - ClockTime) .. " Minutes )"
	end

	if CheckMoon() == "Full Moon" and ClockTime > 5 and ClockTime < 12 then
		return tostring(function6()) .. " ( Fake Moon )"
	end

	if CheckMoon() == "Full Moon" and ClockTime > 12 and ClockTime < 18 then
		return tostring(function6()) .. " ( Will Full Moon In " .. math.floor(18 - ClockTime) .. " Minutes )"
	end

	if CheckMoon() == "Full Moon" and ClockTime > 18 and ClockTime <= 24 then
		return tostring(function6()) .. " ( Will End Moon In " .. math.floor(30 - ClockTime) .. " Minutes )"
	end

	if CheckMoon() == "Next Night" and ClockTime < 12 then
		return tostring(function6()) .. " ( Will Full Moon In " .. math.floor(18 - ClockTime) .. " Minutes )"
	end

	if CheckMoon() == "Next Night" and ClockTime > 12 then
		return tostring(function6()) .. " ( Will Full Moon In " .. math.floor(30 - ClockTime) .. " Minutes )"
	end

	return tostring(function6())
end

	local t8 = {}

				table.insert(t8, {
					v1,
					v1.HumanoidRootPart
				})
			end
		end
	end

	local next_2 = next
	local players, v419 = game:GetService("Players"):GetPlayers()

	for _, v1 in next_2, players, v419 do
		if v1 ~= LocalPlayer and v1.Character and IsAlive(v1.Character) then
			if GetDistance(v1.Character.HumanoidRootPart) <= ((CheckBuddha() == "Buddha V1" or CheckBuddha() == "Buddha V2") and 350 or 60) then
				table.insert(t8, {
					v1.Character,
					v1.Character.HumanoidRootPart
				})
			end
		end
	end

	local next_3 = next
	local children2, v424 = workspace.SeaBeasts:GetChildren()

	for _, v1 in next_3, children2, v424 do
		if v1 and v1:FindFirstChild("HumanoidRootPart") and GetDistance(v1.HumanoidRootPart) <= 700 then
			table.insert(t8, {
				v1,
				v1.HumanoidRootPart
			})
		end
	end

	return t8
end

FastAttack = function()
	pcall(function()
		if thangtat("Character") then
			if LocalPlayer.Character:FindFirstChildOfClass("Tool") and not CheckStun() then
				local v434 = khongbietdattenlagi()

				if not v434 or #v434 == 0 then
					return
				end

				main = v434[1][2]
				RERegisterAttack:FireServer(-math.huge)

				if not Config["No Attack Animation"] then
					pcall(function()
						if CheckWeapon("Melee", "Sword") or CheckWeaponFruit() then
							local v435 = WeaponData[CombatUtil:GetWeaponName(LocalPlayer.Character:FindFirstChildOfClass("Tool"))] or WeaponData[string.lower(CombatUtil:GetWeaponName(LocalPlayer.Character:FindFirstChildOfClass("Tool")))] or WeaponData[CombatUtil:GetPureWeaponName(CombatUtil:GetWeaponName(LocalPlayer.Character:FindFirstChildOfClass("Tool")))]

							if v435 and v435.Moveset and v435.Moveset.Basic then
								local v436 = v435.Moveset.Basic[math.random(1, #v435.Moveset.Basic)]

								if v436 and v436.AnimationId then
									local Animation = Instance.new("Animation")

									Animation.AnimationId = v436.AnimationId
									LocalPlayer.Character:FindFirstChildWhichIsA("Humanoid"):LoadAnimation(Animation):Play()
								end
							end
						end
					end)
					v433:Start()
					v433:Shake(CameraShaker.Presets.Bump)
					pcall(function()
						if CheckWeapon("Melee", "Sword") or CheckWeaponFruit() then
							CombatUtil:AttackStart(main, 1)
							CombatUtil:RunHitDetection(main.Parent or main, 1, {
								_Object = {
									Length = 0.36,
									IsPlaying = true
								}
							})
						end
					end)
				end

				if not CheckWeapon("Blox Fruit") or checkTool("Light-Light") or checkTool("Ice-Ice") then
					if SendHitsToServer then
						SendHitsToServer(main, v434)
					end

					RERegisterHit:FireServer(checkTool("Tool"), v434)
				else
					local LeftClickRemote = LocalPlayer.Character:FindFirstChildOfClass("Tool"):FindFirstChild("LeftClickRemote")

					if LeftClickRemote and LeftClickRemote:IsA("RemoteEvent") then
						LeftClickRemote:FireServer(((main.Position or Vector3.new(main.Position.X, 0, main.Position.Z)) - (LocalPlayer.Character:GetPivot().Position or Vector3.new(LocalPlayer.Character:GetPivot().Position.X, 0, LocalPlayer.Character:GetPivot().Position.Z))).Unit, 1, true)
						LeftClickRemote:FireServer(false)
					end
				end
			end
		end
	end)
end
