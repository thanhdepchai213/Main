	description = "Click to buy the selected Gun",
	callback = function()
    local value37 = t2.value37

    if value37 == "Kabucha" then
        t2.value15:InvokeServer("BlackbeardReward", "Slingshot", "1")
        t2.value15:InvokeServer("BlackbeardReward", "Slingshot", "2")

        return
    end

    if value37 == "Bizarre Rifle" then
        t2.value15:InvokeServer("Ectoplasm", "Buy", 1)

        return
    end

    t2.value15:InvokeServer("BuyItem", value37)
end
})
