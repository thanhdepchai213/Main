local tinsert, tconcat =
	table.insert, function(t)
		local n = #t
		if n == 0 then
			return ""
		end
		while n > 1 do
			local m = 0
			for i = 1, n - 1, 2 do
				m = m + 1
				t[m] = t[i] .. t[i + 1]
			end
			if n % 2 == 1 then
				m = m + 1
				t[m] = t[n]
			end
			for i = m + 1, n do
				t[i] = nil
			end
			n = m
		end
		return t[1]
	end

local function sha256(msg)
	local h0, h1, h2, h3 = 1779033703, 3144134277, 1013904242, 2773480762
	local h4, h5, h6, h7 = 1359893119, 2600822924, 528734635, 1541459225
	local bitlen = #msg * 8
	msg = msg .. "\128"
	while (#msg % 64) ~= 56 do
		msg = msg .. "\0"
	end
	local function be32(x)
		return schar(band(rshift(x, 24), 255), band(rshift(x, 16), 255), band(rshift(x, 8), 255), band(x, 255))
	end
	msg = msg .. be32(floor(bitlen / 4294967296)) .. be32(bitlen % 4294967296)
	local w = {}
	for chunk = 1, #msg, 64 do
		for i = 0, 15 do
			local j = chunk + i * 4
			local b1, b2, b3, b4 = sbyte(msg, j, j + 3)
			w[i] = bor(lshift(b1, 24), lshift(b2, 16), lshift(b3, 8), b4)
		end
		for i = 16, 63 do
			local x, y = w[i - 15], w[i - 2]
			local s0 = bxor(rrotate(x, 7), rrotate(x, 18), rshift(x, 3))
			local s1 = bxor(rrotate(y, 17), rrotate(y, 19), rshift(y, 10))
			w[i] = band(w[i - 16] + s0 + w[i - 7] + s1, MASK)
		end
		local a, b, c, d, e, f, g, h = h0, h1, h2, h3, h4, h5, h6, h7
		for i = 0, 63 do
			local S1 = bxor(rrotate(e, 6), rrotate(e, 11), rrotate(e, 25))
			local ch = bxor(band(e, f), band(bit.bnot(e), g))
			local t1 = band(h + S1 + ch + K[i + 1] + w[i], MASK)
			local S0 = bxor(rrotate(a, 2), rrotate(a, 13), rrotate(a, 22))
			local maj = bxor(band(a, b), band(a, c), band(b, c))
			local t2 = band(S0 + maj, MASK)
			h = g
			g = f
			f = e
			e = add32(d, t1)
			d = c
			c = b
			b = a
			a = add32(t1, t2)
		end
		h0 = add32(h0, a)
		h1 = add32(h1, b)
		h2 = add32(h2, c)
		h3 = add32(h3, d)
		h4 = add32(h4, e)
		h5 = add32(h5, f)
		h6 = add32(h6, g)
		h7 = add32(h7, h)
	end
	local function be(x)
		return schar(band(rshift(x, 24), 255), band(rshift(x, 16), 255), band(rshift(x, 8), 255), band(x, 255))
	end
	return be(h0) .. be(h1) .. be(h2) .. be(h3) .. be(h4) .. be(h5) .. be(h6) .. be(h7)
end
local function u32le(s, i)
	local b1, b2, b3, b4 = sbyte(s, i, i + 3)
	return bor(b1, lshift(b2, 8), lshift(b3, 16), lshift(b4, 24))
end
local function qr(s, a, b, c, d)
	s[a] = add32(s[a], s[b])
	s[d] = lrotate(bxor(s[d], s[a]), 16)
	s[c] = add32(s[c], s[d])
	s[b] = lrotate(bxor(s[b], s[c]), 12)
	s[a] = add32(s[a], s[b])
	s[d] = lrotate(bxor(s[d], s[a]), 8)
	s[c] = add32(s[c], s[d])
	s[b] = lrotate(bxor(s[b], s[c]), 7)
end

local function chacha20(key, nonce, data, initialCounter)
	local keyW = {}
	for i = 1, 8 do
		keyW[i] = u32le(key, (i - 1) * 4 + 1)
	end
	local nonceW = {}
	for i = 1, 3 do
		nonceW[i] = u32le(nonce, (i - 1) * 4 + 1)
	end
	local out, dlen = {}, #data
	local counter = initialCounter or 0
	for off = 0, dlen - 1, 64 do
		local ks = chachaBlock(keyW, counter, nonceW)
		local n = min(64, dlen - off)
		local seg = {}
		for i = 1, n do
			seg[i] = schar(bxor(sbyte(data, off + i), sbyte(ks, i)))
		end
		tinsert(out, tconcat(seg))
		counter = band(counter + 1, MASK)
	end
	return tconcat(out)
end
local B64 = "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"
local B64R = {}
for i = 1, #B64 do
	B64R[sbyte(B64, i)] = i - 1
end
local function b64encode(data)
	local out = {}
	for i = 1, #data, 3 do
		local b1, b2, b3 = sbyte(data, i), sbyte(data, i + 1), sbyte(data, i + 2)
		local n1 = rshift(b1, 2)
		local n2 = bor(lshift(band(b1, 3), 4), b2 and rshift(b2, 4) or 0)
		local n3 = b2 and bor(lshift(band(b2, 15), 2), b3 and rshift(b3, 6) or 0) or nil
		local n4 = b3 and band(b3, 63) or nil
		tinsert(out, ssub(B64, n1 + 1, n1 + 1))
		tinsert(out, ssub(B64, n2 + 1, n2 + 1))
		tinsert(out, n3 and ssub(B64, n3 + 1, n3 + 1) or "=")
		tinsert(out, n4 and ssub(B64, n4 + 1, n4 + 1) or "=")
	end
	return tconcat(out)
end
local function b64decode(str)
	str = str:gsub("[^A-Za-z0-9+/]", "")
	local out, i, n = {}, 1, #str
	while i <= n do
		local c1 = B64R[sbyte(str, i)] or 0
		local c2 = B64R[sbyte(str, i + 1)] or 0
		local c3 = B64R[sbyte(str, i + 2)]
		local c4 = B64R[sbyte(str, i + 3)]
		tinsert(out, schar(bor(lshift(c1, 2), rshift(c2, 4))))
		if c3 then
			tinsert(out, schar(band(bor(lshift(c2, 4), rshift(c3, 2)), 255)))
		end
		if c4 then
			tinsert(out, schar(band(bor(lshift(c3, 6), c4), 255)))
		end
		i = i + 4
	end
	return tconcat(out)
end
local function randomBytes(len)
	local out = {}
	for i = 1, len do
		out[i] = schar(math.random(0, 255))
	end
	return tconcat(out)
end
local function normalizeKey(key)
	if #key == 32 then
		return key
	end
	return sha256(key)
end
function encrypt(key, plaintext)
	local k = normalizeKey(key)
	local nonce = randomBytes(12)
	local ct = chacha20(k, nonce, plaintext)
	return b64encode(nonce .. ct)
end
function decrypt(key, packed)
	local k = normalizeKey(key)
	local raw = b64decode(packed)
	local nonce = ssub(raw, 1, 12)
	local ct = ssub(raw, 13)
	return chacha20(k, nonce, ct)
end

local lastTravel = 0
function smartTween(t)
	if PLACE_ID.sea3() then
		if not checkItem("Valkyrie Helm") then
			return
		end
	end
	if typeof(t) == "CFrame" then
		t = t.Position
	end
	while not character or not character.PrimaryPart or not character:FindFirstChild("Humanoid") do
		task.wait()
	end
	local _is
	if PLACE_ID.sea3() then
		_is = {
			Vector3.new(5700, 1015, -215),
			Vector3.new(-12550, 340, -7500),
			Vector3.new(-5000, 350, -3035),
		}
	elseif PLACE_ID.sea2() then
		_is = {
			Vector3.new(-390, 332, 673),
			Vector3.new(2285, 15, 905),
			Vector3.new(923, 126, 32852),
			Vector3.new(-6509, 83, -133),
		}
	elseif PLACE_ID.sea1() then
		if plr:GetAttribute("CurrentLocation") == "Underwater City" and plr:DistanceFromCharacter(t) >= 3000 then
			game:GetService("ReplicatedStorage").Remotes.CommF_
				:InvokeServer("requestEntrance", Vector3.new(3864, 6, -1926))
			return
		end
		_is = {
			Vector3.new(61163, 11, 1819),
			Vector3.new(-4650, 872, -1775),
			Vector3.new(-7900, 5578, -520),
		}
	else
		return
	end
	local min = nil
	local pos = 1
	for i, v in next, _is do
		if min == nil then
			min = (t - v).Magnitude
		else
			if (t - v).Magnitude < min then
				min = (t - v).Magnitude
				pos = i
			end
		end
	end
	local dist = plr:DistanceFromCharacter(t)
	if min < dist and dist >= 1500 and (tick() - lastTravel) >= 1 then
		warn("Fast travel to ", tostring(_is[pos]), "dist :", dist)
		game.ReplicatedStorage.Remotes.CommF_:InvokeServer("requestEntrance", _is[pos])
		task.wait(0.5)
	end
end

function tween(v, speed, high)
	if speed > 280 and speed < math.huge then
		speed = 280
	end
	while not character or not character.PrimaryPart or not character:FindFirstChild("Humanoid") do
		task.wait()
	end
	if plr.Character and plr.Character:FindFirstChild("HumanoidRootPart") then
		character.Humanoid.Sit = false
		smartTween(v)
		local rootPart = plr.Character:FindFirstChild("HumanoidRootPart")
		if typeof(v) == "Vector3" then
			v = CFrame.new(v)
		end
		local pos = v.Position + Vector3.new(0, high or 0, 0)
		local rot = rootPart.CFrame - rootPart.Position
		v = CFrame.new(pos) * rot
		local distance = (rootPart.Position - v.Position).Magnitude
		local time = distance / speed
		local tweeninfo = TweenInfo.new(time, Enum.EasingStyle.Linear, Enum.EasingDirection.Out)
		tw = tweens:Create(rootPart, tweeninfo, {
			CFrame = v,
		})
		tw:Play()
		return tw
	end
end

function Hop(k, W)
	local check_isvalid = false
	local h = GetServers()
	local X = {}
	for w, D in h do
		table.insert(X, {
			JobId = w,
			Players = D.Count,
			LastUpdate = D.__LastUpdate,
			Region = D.Region,
		})
	end
	print(#X, "servers received")
	for h = 1, #X do
		while task.wait() do
			local h = math.random(1, #X)
			ServerData = X[h]
			if ServerData then
				if not k or ServerData.Players < k then
					if not W or ServerData.Regoin == W then
						print(
							"Found Server:",
							ServerData.JobId,
							"Player Count:",
							ServerData.Players,
							"Region:",
							ServerData.Region
						)
						break
					else
						check_isvalid = true
					end
				end
			end
		end
	end
	if not check_isvalid then
		print("Teleporting to", ServerData.JobId, "..")
		game:GetService("ReplicatedStorage"):WaitForChild("__ServerBrowser"):InvokeServer("teleport", ServerData.JobId)
	else
		local servers = {}
		local ok, req = pcall(function()
			return game:HttpGet(
				"https://games.roblox.com/v1/games/"
					.. game.PlaceId
					.. "/servers/Public?sortOrder=Asc&limit=100&excludeFullGames=true"
			)
		end)
		if not ok or not req then
			return
		end
		local ok2, body = pcall(function()
			return game:GetService("HttpService"):JSONDecode(req)
		end)
		if not ok2 or not body or not body.data then
			return
		end
		for _, v in next, body.data do
			if
				type(v) == "table"
				and tonumber(v.playing)
				and tonumber(v.maxPlayers)
				and v.playing < 2
				and v.id ~= game.JobId
			then
				table.insert(servers, 1, v.id)
			end
		end
		if #servers > 0 then
			game.ReplicatedStorage.__ServerBrowser:InvokeServer("teleport", servers[math.random(1, #servers)])
		end
	end
end

function GetLastQuest() -- -> string
	local data = require(game.ReplicatedStorage.GuideModule).Data

			return i

function GetBestQuest()
	local bestLevelReq = -1
	local bestRnq, bestTaskName, bestRidq = nil, nil, nil
	for rnq, v in next, QuestController.CachedQuestData do
		if table.find({
			"BartiloQuest",
			"Trainees",
			"MarineQuest",
			"CitizenQuest",
		}, tostring(rnq)) then
			continue
		end
		for ridq, ct in next, v do
			if ct.LevelReq < 0 or plr.Data.Level.Value < ct.LevelReq then
				continue
			end
			for O, taskValue in next, ct.Task do
				if taskValue > 1 and ct.LevelReq > bestLevelReq then
					bestLevelReq = ct.LevelReq
					bestRnq = rnq
					bestTaskName = tostring(O)
					bestRidq = ridq
				end
			end
		end
	end
	if not bestRnq then
		return
	end
	return bestLevelReq, bestRnq, bestTaskName, bestRidq
end

function GetQuest()
	local Level = plr.Data.Level.Value
	local msn, nq, idq
	local min = 0
	if Level >= 275 and Level < 300 then
		nq, msn, idq = "ColosseumQuest", "Togga Warrior", 1
	elseif Level >= 1450 and PLACE_ID.sea2() then
		nq, msn, idq = "ForgottenQuest", "Water Fighter", 2
	elseif Level >= 700 and PLACE_ID.sea1() then
		nq, msn, idq = "FountainQuest", "Galley Captain", 2
	else
		min, nq, msn, idq = GetBestQuest()
	end
	local q = {}
	if Level >= 10 and not GetQuestData() and GetLastQuest() == msn then
		for rnq, v in next, QuestController.CachedQuestData do
			for _, N in next, v do
				for x, _ in next, N.Task do
					if tostring(x) == msn then
						for iq, v2 in next, v do
							for n, ct in next, v2.Task do
								if n ~= msn and ct > 1 then
									q.NameMonster = v2.LevelReq <= Level and tostring(n) or msn
									q.NameQuest = v2.LevelReq <= Level and rnq or nq
									q.ID = v2.LevelReq <= Level and iq or idq
									return q -- Double Quest or not
								end
							end
						end
					end
				end
			end
		end
	end
	q.NameMonster = msn
	q.NameQuest = nq
	q.ID = idq
	return q
end

FarmMaterial = function(material)
	getMaterialMon(material)
	local mob = nil
	local min = math.huge
	for _, v in next, workspace.Enemies:GetChildren() do
		if v.PrimaryPart and v:FindFirstChild("Humanoid") and v.Humanoid.Health > 0 and table.find(materialMon, v.Name) then
			local dist = plr:DistanceFromCharacter(v.PrimaryPart.Position)
			if dist < min then
				min = dist
				mob = v
			end
		end
	end
	if mob and mob.PrimaryPart and mob.Humanoid and mob.Humanoid.Health > 0 then
		repeat
			task.wait()
			equipWeapon()
			tween(mob.PrimaryPart.CFrame * CFrame.new(0, 20, 10), 350)
			if plr:DistanceFromCharacter(mob.PrimaryPart.CFrame.Position) <= 200 then
				tween(mob.PrimaryPart.CFrame * CFrame.new(0, 20, 10), math.huge)
				if plr:DistanceFromCharacter(mob.PrimaryPart.CFrame.Position) <= 45 then
					bringMob(mob.Name)
					FastAttack()
				end
			end
		until not mob or not mob.PrimaryPart or not mob:FindFirstChild("Humanoid") or mob.Humanoid.Health <= 0
	else
		tween(materialPos, 350).Completed:Wait()
	end
end

AutoDungeon = function()
	SetStatus("Doing Raid")
	warn("now do raid")
	local res = game.ReplicatedStorage.Remotes.CommF_:InvokeServer("RaidsNpc", "Select", "Dark")
	while not character or not character.PrimaryPart do
		task.wait()
	end
	if not RaidController.isRaiding then
		if not (plr.Backpack:FindFirstChild("Special Microchip") or character:FindFirstChild("Special Microchip")) then
			if
				plr.Data.Beli.Value >= 100000
				and ((type(res) == "number" and res == 1) or (type(res) == "string" and not res:find("wait")))
			then
				game.ReplicatedStorage.Remotes.CommF_:InvokeServer("RaidsNpc", "Select", "Dark")
			else
				PAUSE_AUTO_STORE = true
				game.ReplicatedStorage.Remotes.CommF_:InvokeServer("LoadFruit", getUseableFruit())
				task.wait(0.5)
				game.ReplicatedStorage.Remotes.CommF_:InvokeServer("RaidsNpc", "Select", "Dark")
				PAUSE_AUTO_STORE = false
			end
		end
		if PLACE_ID.sea2() then
			local clickDetector = workspace.Map.CircleIsland.RaidSummon2:FindFirstChildWhichIsA("ClickDetector", true)
			if clickDetector then
				pcall(fireclickdetector, clickDetector)
			end
		elseif PLACE_ID.sea3() then
			tween(Vector3.new(-5018, 315, -2828), 300).Completed:Wait()
			task.wait(1)
			local clickDetector = workspace.Map["Boat Castle"].RaidSummon2:FindFirstChildWhichIsA("ClickDetector", true)
			if clickDetector then
				pcall(fireclickdetector, clickDetector)
			end
		end
		local startWait = tick()
		repeat
			SetStatus("Joining for raid")
			StopTween()
			task.wait(0.25)
		until RaidController.isRaiding or (tick() - startWait) >= 30
		task.wait(0.5)
		while RaidController.isRaiding do
			SetStatus("Clear Raid")
			for _, v in next, workspace.Enemies:GetChildren() do
				if
					v.PrimaryPart
					and v:FindFirstChildOfClass("Humanoid")
					and v:FindFirstChildOfClass("Humanoid").Health > 0
					and plr:DistanceFromCharacter(v.PrimaryPart.Position) <= 2000
				then
					repeat
						task.wait()
						pcall(sethiddenproperty, plr, "SimulationRadius", math.huge)
						pcall(function()
							tween(v.PrimaryPart.Position + Vector3.new(0, 28, 15), 350)
							if plr:DistanceFromCharacter(v.PrimaryPart.Position) <= 200 then
								tween(v.PrimaryPart.Position + Vector3.new(0, 28, 15), math.huge)
								if plr:DistanceFromCharacter(v.PrimaryPart.Position) <= 50 then
									equipWeapon()
									FastAttack()
								end
							end
						end)
					until not v
						or not v.PrimaryPart
						or not v:FindFirstChildOfClass("Humanoid")
						or v:FindFirstChildOfClass("Humanoid").Health == 0
					break
				end
			end
			local found = false
			for i = 5, 1, -1 do
				for _, z in next, workspace["_WorldOrigin"].Locations:GetChildren() do
					if
						z.Name == ("Island " .. i)
						and plr:DistanceFromCharacter(
								workspace["_WorldOrigin"].Locations["Island " .. i]:GetPivot().Position
							)
							<= 6000
					then
						found = true
						tween(
							workspace["_WorldOrigin"].Locations["Island " .. i]:GetPivot() * CFrame.new(100, 70, 100),
							350
						)
						task.wait(0.5)
						break
					end
				end
				if found then
					break
				end
			end
			task.wait()
		end
	end
	task.wait(5)
end

local c = 0
function AutoChest()
	local m = workspace.Map:GetDescendants()
	local h = character:FindFirstChild("Humanoid") or false
	for i = 1, #m do
		local v = m[i]
		if
			v:IsA("BasePart")
			and v.Name:find("Chest")
			and (v.Name:match("%d+") or v.Name:find("Frag"))
			and v.CanTouch
		then
			if
				(function()
					for _, v in
						{
							character,
							plr.Backpack,
						}
					do
						for _, v2 in next, v:GetChildren() do
							if
								v2.Name == "God's Chalice"
								or v2.Name == "Fist of Darkness"
								or v2.Name == "Sweet Chalice"
							then
								return true
							end
						end
					end
					return false
				end)()
			then
				break
			end
			if c >= 10 then
				c = 0
				h.Health = 0
			else
				if character.PrimaryPart and character:FindFirstChildOfClass("Humanoid") then
					if character.Humanoid.Health > 0 then
						character:SetPrimaryPartCFrame(v.CFrame * CFrame.new(0, 2, 0))
						c = c + 1
					end
				end
			end
			task.wait(0.2)
		end
	end
end

function CheckHakiLegendInTitles(NAME_HAKI)
	local ok, ngu = pcall(function()
		game.ReplicatedStorage.Remotes.CommF_:InvokeServer("getTitles")
	end)
	if not ok or not ngu or typeof(ngu) ~= "table" then
		return false
	end
	local gg = ok[NAME_HAKI]
	if not gg then
		return false
	end
	for _, v in pairs(ngu) do
		if v.Unlocked and v.Description and v.Description:find(gg) then
			return true
		end
	end
	return false
end

function DispTime(W, a)
	W = tonumber(W)
	if not W then
		return "[err]"
	end
	local h = math.floor(W / 86400)
	local X = math.floor(math.fmod(W, 86400) / 3600)
	local w = math.floor(math.fmod(W, 3600) / 60)
	local D = math.floor(math.fmod(W, 60))
	if a then
		return (h .. "day, " .. X .. "hrs, " .. w .. "min, " .. D .. "sec.")
	end
	return (h .. "day, " .. X .. "hrs.")
end

local SessionStartTime = os.time()
local OldSessionTime = 0
local LastWrittenTime = 0
pcall(function()
	if isfile and readfile and isfile(".tdif-" .. game.Players.LocalPlayer.Name) then
		OldSessionTime = tonumber(readfile(".tdif-" .. game.Players.LocalPlayer.Name)) or 0
	end
end)

task.spawn(function()
	while true do
		if SetText and DispTime then
			local J = os.time() - SessionStartTime
			local r = J + OldSessionTime
			if r ~= LastWrittenTime then
				LastWrittenTime = r
				pcall(function()
					writefile(".tdif-" .. game.Players.LocalPlayer.Name, tostring(r))
				end)
			end
			SetText("LiveTime", "Total Elapsed Time: " .. DispTime(r, true) .. " Elapsed Time: " .. DispTime(J, true))
		end
		task.wait(1)
	end
end)
